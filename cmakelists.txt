# ------------------------------------------------------------
# Place executables in a dedicated output directory
# ------------------------------------------------------------
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY
    ${CMAKE_BINARY_DIR}/../bin
)

set(CMAKE_LIBRARY_OUTPUT_DIRECTORY
    ${CMAKE_BINARY_DIR}/../bin
)

set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY
    ${CMAKE_BINARY_DIR}/../lib
)

cmake_minimum_required(VERSION 3.10)
project(Voxels)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# ------------------------------------------------------------------
# Compile Slang shaders directly into C headers using slangc
# ------------------------------------------------------------------
find_program(SLANGC
    NAMES slangc
    HINTS ENV VULKAN_SDK ENV SLANG_SDK
    PATH_SUFFIXES bin
    REQUIRED
)

file(GLOB_RECURSE SLANG_SRC_FILES
    "${CMAKE_SOURCE_DIR}/src/shaders/slang/*.slang"
)

set(SHADER_HEADER_DIR ${CMAKE_SOURCE_DIR}/src/shaders)
file(MAKE_DIRECTORY ${SHADER_HEADER_DIR})

set(SHADER_HEADERS)
foreach(SLANG_SRC ${SLANG_SRC_FILES})
    get_filename_component(FILE_NAME ${SLANG_SRC} NAME_WE)
    set(HDR ${SHADER_HEADER_DIR}/${FILE_NAME}.h)

    # Variable name inside the header
    set(VAR_NAME ${FILE_NAME}_spirv)

    add_custom_command(
        OUTPUT ${HDR}
        COMMAND ${SLANGC}
            ${SLANG_SRC}
            -target spirv
            -profile spirv_1_3
            -emit-spirv-directly
            -source-embed-style u32
            -source-embed-name ${VAR_NAME}
            -o ${HDR}
        DEPENDS ${SLANG_SRC}
        COMMENT "Compiling ${SLANG_SRC} -> ${HDR}"
        VERBATIM
    )

    list(APPEND SHADER_HEADERS ${HDR})
endforeach()

add_custom_target(Shaders ALL DEPENDS ${SHADER_HEADERS})

# ------------------------------------------------------------------
# Build the program
# ------------------------------------------------------------------
enable_language(C)
# Core source (recursive)
file(GLOB_RECURSE SRC_FILES
    "${CMAKE_SOURCE_DIR}/src/*.cpp"
    "${CMAKE_SOURCE_DIR}/src/*.c"
)

# ImGui (non-recursive where needed)
file(GLOB IMGUI_FILES
    "${CMAKE_SOURCE_DIR}/imgui/*.cpp"
    "${CMAKE_SOURCE_DIR}/imgui/backends/*.cpp"
)

# Combine them
set(ALL_SOURCES
    ${SRC_FILES}
    ${IMGUI_FILES}
)

add_executable(Voxels ${ALL_SOURCES})

find_package(Vulkan REQUIRED)

target_include_directories(Voxels PRIVATE
    ${CMAKE_SOURCE_DIR}/include
    ${CMAKE_SOURCE_DIR}/src/
    ${CMAKE_SOURCE_DIR}/src/core
    ${CMAKE_SOURCE_DIR}/imgui
)

target_link_directories(Voxels PRIVATE ${CMAKE_SOURCE_DIR}/lib/SDL3)

target_link_libraries(Voxels
    PRIVATE
    Vulkan::Vulkan
    SDL3
)

target_link_options(Voxels PRIVATE
    -static-libgcc
    -static-libstdc++
)

# Ensure shaders are built first
add_dependencies(Voxels Shaders)

message(STATUS "Found slang sources: ${SLANG_SRC_FILES}")
